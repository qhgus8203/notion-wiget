  <script>

    // =========================================================================

    // [커스텀 설정 영역]

    // 1. 배치 방향 설정: 가로로 나열하려면 'row', 세로로 쌓으려면 'column' 입력

    // =========================================================================

    const LAYOUT_DIRECTION = 'row'; 


    // =========================================================================

    // 2. 디데이 항목 추가 및 수정

    // - title: 그래프 아래에 표시될 디데이 이름

    // - startDate: 진행률(퍼센트) 계산의 시작 기준일 (YYYY-MM-DD)

    // - targetDate: 목표 디데이 날짜 (YYYY-MM-DD)

    // - color: 원형 선 그래프의 포인트 컬러 (HEX 코드)

    // =========================================================================

    const DDAY_ITEMS = [

      {

        title: "기말고사",

        startDate: "2026-09-01",

        targetDate: "2026-12-15",

        color: "#2563eb" // 블루

      },

      {

        title: "2027 새해",

        startDate: "2026-01-01",

        targetDate: "2027-01-01",

        color: "#10b981" // 에메랄드 그린

      }

      // 새로운 항목을 추가하려면 위 { ... } 블록을 복사해서 쉼표(,) 뒤에 붙여넣으세요!

    ];

    // =========================================================================


    function renderDDays() {

      const listEl = document.getElementById('dday-list');

      listEl.style.flexDirection = LAYOUT_DIRECTION;

      listEl.innerHTML = '';


      const today = new Date();

      today.setHours(0, 0, 0, 0);


      const radius = 41;

      const circumference = 2 * Math.PI * radius;


      DDAY_ITEMS.forEach(item => {

        const start = new Date(item.startDate + "T00:00:00");

        const target = new Date(item.targetDate + "T00:00:00");


        // 남은 일수 계산

        const diffTime = target - today;

        const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));


        // "D-" 없이 숫자만 표시 (지났을 경우 +숫자)

        let ddayLabel = "";

        if (diffDays === 0) {

          ddayLabel = "0";

        } else if (diffDays > 0) {

          ddayLabel = `${diffDays}`;

        } else {

          ddayLabel = `+${Math.abs(diffDays)}`;

        }


        // 퍼센티지 계산

        const totalDuration = target - start;

        const elapsedDuration = today - start;

        let percentage = 0;


        if (totalDuration <= 0 || diffDays <= 0) {

          percentage = 100;

        } else if (elapsedDuration <= 0) {

          percentage = 0;

        } else {

          percentage = Math.min(Math.max((elapsedDuration / totalDuration) * 100, 0), 100);

        }


        const strokeDashoffset = circumference - (percentage / 100) * circumference;


        const itemEl = document.createElement('div');

        itemEl.className = 'dday-item';

        itemEl.innerHTML = `

          <div class="circle-wrapper" title="진행률 ${Math.round(percentage)}%">

            <svg class="progress-ring" width="96" height="96">

              <circle class="progress-ring__bg" r="${radius}" cx="48" cy="48" />

              <circle 

                class="progress-ring__circle" 

                r="${radius}" 

                cx="48" 

                cy="48" 

                style="stroke: ${item.color}; stroke-dasharray: ${circumference} ${circumference}; stroke-dashoffset: ${strokeDashoffset};"

              />

            </svg>

            <span class="dday-count">${ddayLabel}</span>

          </div>

          <div class="dday-name">${item.title}</div>

        `;


        listEl.appendChild(itemEl);

      });

    }


    renderDDays();

  </script>
