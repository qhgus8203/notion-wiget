    <div class="dday-list" id="dday-list"></div>
  </div>
  <script>
    const LAYOUT_DIRECTION = 'row'; 
    const DDAY_ITEMS = [
      { title: "기말고사", startDate: "2026-09-01", targetDate: "2026-12-15", color: "#2563eb" },
      { title: "2027 새해", startDate: "2026-01-01", targetDate: "2027-01-01", color: "#10b981" }
    ];
    function renderDDays() {
      const listEl = document.getElementById('dday-list');
      listEl.style.flexDirection = LAYOUT_DIRECTION;
      listEl.innerHTML = '';
      const today = new Date();
      today.setHours(0, 0, 0, 0);
      const radius = 41;
      const circumference = 2 * Math.PI * radius;
      DDAY_ITEMS.forEach(item => {
        const start = new Date(item.startDate + "T00:00:00");
        const target = new Date(item.targetDate + "T00:00:00");
        const diffTime = target - today;
        const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
        let ddayLabel = "";
        if (diffDays === 0) { ddayLabel = "0"; }
        else if (diffDays > 0) { ddayLabel = `${diffDays}`; }
        else { ddayLabel = `+${Math.abs(diffDays)}`; }
        const totalDuration = target - start;
        const elapsedDuration = today - start;
        let percentage = 0;
        if (totalDuration <= 0 || diffDays <= 0) { percentage = 100; }
        else if (elapsedDuration <= 0) { percentage = 0; }
        else { percentage = Math.min(Math.max((elapsedDuration / totalDuration) * 100, 0), 100); }
        const strokeDashoffset = circumference - (percentage / 100) * circumference;
        const itemEl = document.createElement('div');
        itemEl.className = 'dday-item';
        itemEl.innerHTML = `
          <div class="circle-wrapper" title="진행률 ${Math.round(percentage)}%">
            <svg class="progress-ring" width="96" height="96">
              <circle class="progress-ring__bg" r="${radius}" cx="48" cy="48" />
              <circle class="progress-ring__circle" r="${radius}" cx="48" cy="48" style="stroke: ${item.color}; stroke-dasharray: ${circumference} ${circumference}; stroke-dashoffset: ${strokeDashoffset};" />
            </svg>
            <span class="dday-count">${ddayLabel}</span>
          </div>
          <div class="dday-name">${item.title}</div>
        `;
        listEl.appendChild(itemEl);
      });
    }
    renderDDays();
  </script>
